#!/usr/bin/env bash
# seekdb + hipVS notebook image entrypoint: starts Jupyter with root at the
# mounted /workspace. The observer is started from inside the notebook.
set -e
export LD_LIBRARY_PATH=/seekdb_workspace/bridge:/opt/hipvs/lib:/opt/rocm/lib
mkdir -p /workspace
exec /opt/venv/bin/python3 -m jupyter_server \
     --allow-root --ip=0.0.0.0 --port="${JUPYTER_PORT:-8888}" --no-browser \
     --ServerApp.root_dir=/workspace \
     --ServerApp.allow_remote_access=True \
     --IdentityProvider.token="${JUPYTER_TOKEN:-seekdb-hipvs}"
